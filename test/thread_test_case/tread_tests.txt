class arg

    }
    arg (arg&&) {
        std::cout << "arg::move\n"; 
    }

    arg& operator =(const arg&){
        std::cout << "arg::copy_as\n"; 
        return *this;
    }

    arg& operator =(arg&&){
        std::cout << "arg::move_as\n"; 
        return *this;
    }

    void proc()
    {
        std::cout << "arg::processing\n";
    }

    void task(ufo::token_t){

    }

    ~arg() {
        std::cout << "arg::destructor\n"; 
    }
};

inline void Task(arg&& arg, ufo::token_t token) {
    std::cout << "123\n";
    arg.proc();
    uint16_t cnt = 0;
    if (token)
    {
        std::cout << "token token\n";
    }
    
    while (token)
    {
        if (!token)
        {
            std::cout << "!token\n";
        }
        if (cnt == 5)
        {
            break;
        }
        
        std::cout << "cnt:" << ++cnt << '\n';
        ufo::utl::sleep_for(1000);
        if (token)
        {
            std::cout << "token token\n";
        }
    }
}
